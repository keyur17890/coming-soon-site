# ─────────────────────────────────────────────────────────────────
# GENERAL CRAWLER DIRECTIVES
# ─────────────────────────────────────────────────────────────────

# Allow all crawlers by default
User-agent: *

# Allow crawling of main content
Allow: /

# ─────────────────────────────────────────────────────────────────
# DISALLOWED PATHS
# ─────────────────────────────────────────────────────────────────

# Block admin and system directories
Disallow: /admin/
Disallow: /includes/
Disallow: /config.php
Disallow: /.env
Disallow: /.env.example

# Block form submission endpoints
Disallow: /contact-submit.php
Disallow: /newsletter-subscribe.php

# Block temporary and cache files
Disallow: /tmp/
Disallow: /cache/
Disallow: /logs/

# Block error pages
Disallow: /404.php
Disallow: /500.php

# Block search results pages (if any)
Disallow: /search?
Disallow: /*?s=
Disallow: /*?search=

# Block duplicate content from query parameters
Disallow: /*?utm_*
Disallow: /*?ref=
Disallow: /*?source=
Disallow: /*?campaign=

# Block print and mobile versions
Disallow: /*?print=
Disallow: /*?mobile=

# ─────────────────────────────────────────────────────────────────
# CRAWL RATE LIMITING
# ─────────────────────────────────────────────────────────────────

# Request a 1 second delay between requests (optional, not all crawlers honor this)
Crawl-delay: 1

# ─────────────────────────────────────────────────────────────────
# SPECIFIC CRAWLER RULES

User-agent: Googlebot
Allow: /

Allow: /assets/images/
Disallow: /admin/

User-agent: Bingbot

# Yandex Bot

# BLOCK BAD BOTS
